Example 1: single PyRanges object. Tap the per-panel view dropdown. Current prototype names are still squish / packed / list / zip.
import pyranges1 as pr
import pyrangeyes as pre

pre.set_engine("ply")

data = pr.PyRanges({
    "Chromosome": ["chr1"] * 7,
    "Start": [100, 106, 112, 118, 124, 130, 136],
    "End": [104, 110, 116, 122, 128, 134, 140],
    "id": [f"tx{i}" for i in range(1, 8)],
    "label": [f"mobile_demo_{i}" for i in range(1, 8)],
})

fig = pre.browser(
    data,
    id_col="id",
    text={"label": "label", "position": "right", "avoid_overlaps": False},
    default_mode="packed",
)
fig.show()
