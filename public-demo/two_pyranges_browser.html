Example 2: two PyRanges objects. This exercises the multi-object path; the intended interface will put a compact menu in each object panel/title area.
import pyranges1 as pr
import pyrangeyes as pre

pre.set_engine("ply")

genes = pr.PyRanges({
    "Chromosome": ["chr1"] * 4,
    "Start": [100, 125, 160, 210],
    "End": [118, 150, 195, 250],
    "id": ["geneA", "geneB", "geneC", "geneD"],
})

peaks = pr.PyRanges({
    "Chromosome": ["chr1"] * 5,
    "Start": [104, 132, 170, 218, 238],
    "End": [110, 138, 176, 225, 245],
    "id": ["peak1", "peak2", "peak3", "peak4", "peak5"],
})

fig = pre.browser(
    [genes, peaks],
    id_col="id",
    default_mode="packed",
)
fig.show()
